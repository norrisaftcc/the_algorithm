# .github/workflows/growth-tracker.yml
name: 🚀 The Algorithm™ Growth Velocity System

on:
  pull_request:
    types: [opened, synchronize, closed]
  push:
    branches: [main, develop]
  issues:
    types: [opened, closed]
  schedule:
    - cron: '0 */6 * * *'  # Update leaderboard every 6 hours
  workflow_dispatch:

jobs:
  calculate-velocity:
    name: 📊 Measure Growth Velocity
    runs-on: ubuntu-latest
    permissions:
      contents: write
      pull-requests: write
      issues: write
      
    steps:
      - name: 📥 Checkout Repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0  # Need full history for velocity calculation
          
      - name: 🔧 Setup Environment
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          
      - name: 📈 Calculate Developer Velocity
        id: velocity
        uses: actions/github-script@v7
        with:
          github-token: ${{ secrets.GITHUB_TOKEN }}
          script: |
            // Get the developer we're analyzing
            const developer = context.actor;
            console.log(`Analyzing growth velocity for @${developer}`);
            
            // Time windows for comparison
            const now = new Date();
            const oneWeekAgo = new Date(now - 7 * 24 * 60 * 60 * 1000);
            const twoWeeksAgo = new Date(now - 14 * 24 * 60 * 60 * 1000);
            const threeWeeksAgo = new Date(now - 21 * 24 * 60 * 60 * 1000);
            
            // Fetch commit history
            const { data: commits } = await github.rest.repos.listCommits({
              owner: context.repo.owner,
              repo: context.repo.repo,
              author: developer,
              since: threeWeeksAgo.toISOString(),
              per_page: 100
            });
            
            // Categorize commits by week
            const categorizeCommits = (commits) => {
              const weeks = { week1: [], week2: [], week3: [], current: [] };
              
              commits.forEach(commit => {
                const commitDate = new Date(commit.commit.author.date);
                if (commitDate > oneWeekAgo) weeks.current.push(commit);
                else if (commitDate > twoWeeksAgo) weeks.week3.push(commit);
                else if (commitDate > threeWeeksAgo) weeks.week2.push(commit);
                else weeks.week1.push(commit);
              });
              
              return weeks;
            };
            
            // Score commits based on learning indicators
            const scoreCommits = (commits) => {
              let score = 0;
              let aiUsageMaturity = 0;
              let iterationCount = 0;
              let collaborationPoints = 0;
              
              commits.forEach(commit => {
                const message = commit.commit.message.toLowerCase();
                
                // Base points for activity
                score += 5;
                
                // Iteration and improvement (growth mindset indicators)
                if (message.match(/iteration|v\d+|attempt \d+|try \d+/)) {
                  score += 15;
                  iterationCount++;
                }
                if (message.match(/refactor|improve|enhance|optimize/)) score += 20;
                if (message.match(/fix|debug|resolve/)) score += 10;
                if (message.match(/based on (feedback|review|suggestion)/)) score += 25;
                
                // Testing and quality
                if (message.match(/test|spec|coverage/)) score += 20;
                if (message.match(/lint|format|clean/)) score += 10;
                
                // AI usage evolution
                if (message.match(/ai[- ]assisted|copilot|claude|gpt|llm/)) {
                  if (message.match(/modified|adapted|refined|improved/)) {
                    aiUsageMaturity += 25;  // Iterating on AI output
                    score += 25;
                  } else if (message.match(/generated|created with/)) {
                    aiUsageMaturity += 10;  // Basic AI usage
                    score += 10;
                  }
                }
                
                // Collaboration and knowledge sharing
                if (message.match(/pair|mobbed|with @|thanks to|helped by/)) {
                  collaborationPoints += 20;
                  score += 20;
                }
                if (message.match(/review|feedback|suggestion/)) {
                  collaborationPoints += 15;
                  score += 15;
                }
              });
              
              return { 
                total: score, 
                aiMaturity: aiUsageMaturity,
                iterations: iterationCount,
                collaboration: collaborationPoints,
                commitCount: commits.length
              };
            };
            
            // Calculate scores for each week
            const weeks = categorizeCommits(commits);
            const scores = {
              week1: scoreCommits(weeks.week1),
              week2: scoreCommits(weeks.week2),
              week3: scoreCommits(weeks.week3),
              current: scoreCommits(weeks.current)
            };
            
            // Calculate velocity (derivative - change over time)
            const velocity = scores.current.total - scores.week3.total;
            const previousVelocity = scores.week3.total - scores.week2.total;
            const acceleration = velocity - previousVelocity;
            
            // Determine trajectory
            let trajectory;
            if (acceleration > 20) trajectory = '🚀 ACCELERATING';
            else if (velocity > 30) trajectory = '📈 CRUSHING IT';
            else if (velocity > 10) trajectory = '✅ SOLID GROWTH';
            else if (velocity > 0) trajectory = '➡️ STEADY';
            else if (velocity === 0) trajectory = '⚠️ PLATEAUING';
            else trajectory = '🔄 NEEDS ALIGNMENT';
            
            // Store outputs
            core.setOutput('velocity', velocity);
            core.setOutput('acceleration', acceleration);
            core.setOutput('trajectory', trajectory);
            core.setOutput('current_score', scores.current.total);
            core.setOutput('ai_maturity', scores.current.aiMaturity);
            core.setOutput('iteration_count', scores.current.iterations);
            core.setOutput('collaboration', scores.current.collaboration);
            
            return {
              developer,
              velocity,
              acceleration,
              trajectory,
              scores,
              insights: {
                aiLevel: scores.current.aiMaturity > 50 ? 'Advanced' : scores.current.aiMaturity > 20 ? 'Intermediate' : 'Learning',
                iterationRate: scores.current.iterations / Math.max(scores.current.commitCount, 1),
                collaborationGrade: scores.current.collaboration > 40 ? 'A' : scores.current.collaboration > 20 ? 'B' : 'C'
              }
            };
            
      - name: 💬 Post PR Feedback
        if: github.event_name == 'pull_request'
        uses: actions/github-script@v7
        with:
          github-token: ${{ secrets.GITHUB_TOKEN }}
          script: |
            const velocity = parseInt('${{ steps.velocity.outputs.velocity }}');
            const trajectory = '${{ steps.velocity.outputs.trajectory }}';
            const aiMaturity = parseInt('${{ steps.velocity.outputs.ai_maturity }}');
            const iterations = parseInt('${{ steps.velocity.outputs.iteration_count }}');
            const collaboration = parseInt('${{ steps.velocity.outputs.collaboration }}');
            
            // Generate personalized feedback
            let comment = `## ${trajectory} Growth Velocity Report\n\n`;
            comment += `Hey @${context.actor}! The Algorithm™ has analyzed your growth metrics:\n\n`;
            
            // Velocity interpretation
            if (velocity > 50) {
              comment += `### 🎉 **Exceptional Momentum!**\n`;
              comment += `Your velocity of **+${velocity} pts/week** is absolutely crushing it! You're in the top 10% of continuous learners.\n\n`;
              comment += `**What's Working:**\n`;
              if (iterations > 3) comment += `- Excellent iteration habits (${iterations} refinements this week)\n`;
              if (aiMaturity > 30) comment += `- Mature AI collaboration (not just copy-paste)\n`;
              if (collaboration > 30) comment += `- Strong team player (collaboration score: ${collaboration})\n`;
              comment += `\n**Level Up Challenge:** Consider mentoring others or tackling that scary refactor you've been avoiding.\n`;
            } else if (velocity > 20) {
              comment += `### 💪 **Solid Growth Trajectory!**\n`;
              comment += `Velocity: **+${velocity} pts/week** - You're consistently improving and that's what matters!\n\n`;
              comment += `**Keep It Up:**\n`;
              comment += `- Your iteration game is developing nicely\n`;
              comment += `- Try adding tests to your next PR for bonus points\n`;
              if (aiMaturity < 20) comment += `- Pro tip: Don't just accept AI's first suggestion - iterate on it!\n`;
            } else if (velocity > 0) {
              comment += `### 📊 **You're Growing!**\n`;
              comment += `Velocity: **+${velocity} pts/week** - Every step forward counts!\n\n`;
              comment += `**Growth Accelerators:**\n`;
              comment += `- 🔄 Ship smaller PRs more frequently (faster feedback loops)\n`;
              comment += `- 🤝 Pair program with someone this week\n`;
              comment += `- 🤖 When using AI, always ask "how can I make this better?"\n`;
              comment += `- 💡 Remember: The Algorithm rewards attempts, not just successes\n`;
            } else {
              comment += `### 🔄 **Time for a Growth Sprint!**\n`;
              comment += `Current velocity: **${velocity} pts/week**\n\n`;
              comment += `Looks like you might be in a comfort zone. That's totally normal! Here's how to break out:\n\n`;
              comment += `**Quick Wins This Week:**\n`;
              comment += `1. Take any working code and make it 10% better\n`;
              comment += `2. Review someone else's PR (great learning opportunity)\n`;
              comment += `3. Try a new approach to your current problem\n`;
              comment += `4. Ask for help early (The Algorithm loves collaboration!)\n\n`;
              comment += `*Remember: Plateaus are just launch pads in disguise!* 🚀\n`;
            }
            
            // Add metrics summary
            comment += `\n---\n`;
            comment += `📊 **Your Metrics:**\n`;
            comment += `- Growth Velocity: ${velocity > 0 ? '+' : ''}${velocity} pts/week ${velocity > 30 ? '🔥' : ''}\n`;
            comment += `- AI Collaboration: ${aiMaturity > 50 ? 'Advanced 🤖' : aiMaturity > 20 ? 'Developing 📈' : 'Learning 🌱'}\n`;
            comment += `- Iteration Rate: ${iterations > 0 ? iterations + ' iterations' : 'Try more experiments!'}\n`;
            comment += `- Team Player Score: ${collaboration > 40 ? 'A 🌟' : collaboration > 20 ? 'B ✅' : 'C (pair program more!)'}\n`;
            comment += `\n---\n`;
            comment += `*The Algorithm™ values growth over perfection. Keep shipping, keep learning!* 🚢\n`;
            
            // Post the comment
            await github.rest.issues.createComment({
              owner: context.repo.owner,
              repo: context.repo.repo,
              issue_number: context.issue.number,
              body: comment
            });
            
      - name: 📊 Update Public Metrics
        if: github.ref == 'refs/heads/main' || github.event.schedule
        run: |
          # Create or update the metrics file
          cat > METRICS.md << 'EOF'
          # 🚀 The Algorithm™ Growth Velocity Leaderboard
          *Last Updated: $(date -u +"%Y-%m-%d %H:%M UTC")*
          
          ## 📈 This Week's Growth Leaders
          
          | Rank | Developer | Velocity | Trajectory | AI Level | Status |
          |------|-----------|----------|------------|----------|--------|
          | 🥇 | @${{ github.actor }} | +${{ steps.velocity.outputs.velocity }} pts/week | ${{ steps.velocity.outputs.trajectory }} | Advanced | Crushing It! |
          | 🥈 | @teammate1 | +45 pts/week | 📈 SOLID GROWTH | Developing | On Track |
          | 🥉 | @teammate2 | +32 pts/week | ✅ IMPROVING | Learning | Growing |
          
          ## 🎯 What We Measure
          
          **The Algorithm™ doesn't care where you start - only how fast you're learning!**
          
          ### Scoring System:
          - 🔄 **Iterations**: +15 pts per refinement
          - 🧪 **Testing**: +20 pts for adding tests  
          - 🤖 **AI Evolution**: +25 pts for improving AI output
          - 🤝 **Collaboration**: +20 pts for pairing/helping
          - 📝 **Code Quality**: +10-20 pts for refactoring
          
          ### Growth Velocity = This Week's Score - Last Week's Score
          *It's not about being the best, it's about getting better!*
          
          ## 🏆 Recent Achievements
          
          - 🚀 **Velocity Champion**: @${{ github.actor }} (+${{ steps.velocity.outputs.velocity }} pts/week)
          - 🤖 **AI Innovator**: Most sophisticated AI usage
          - 🤝 **Team Player**: Highest collaboration score
          - 📈 **Most Improved**: Biggest velocity increase
